### 1. 캘린더 그리드 직접 구현
```javascript
function renderCal() {
  const firstDay = new Date(calYear, calMonth - 1, 1);
  const daysInMonth = new Date(calYear, calMonth, 0).getDate();
  const startDow = firstDay.getDay();  // 0=일, 1=월, ..., 6=토

  let cells = '';
  
  // 이전달 빈 칸 채우기
  for (let i = 0; i < startDow; i++) {
    cells += `<div class="cal-day other-month">...</div>`;
  }
  
  // 이번달 날짜 렌더링
  for (let d = 1; d <= daysInMonth; d++) {
    const ds = `${calYear}-${String(calMonth).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const dow = (startDow + d - 1) % 7;  // 요일 계산
    
    // 해당 날짜의 납기 건들 찾기
    const dayItems = items.filter(i => i.dueDate === ds);
    
    cells += `
      <div class="cal-day ${ds === today ? 'today' : ''}">
        <span class="day-no ${dow === 0 ? 'sunday' : dow === 6 ? 'saturday' : ''}">${d}</span>
        ${dayItems.map(item => renderEvent(item)).join('')}
      </div>`;
  }
  
  // CSS Grid 7열로 렌더링
  document.getElementById('calGrid').innerHTML = cells;
}
```

let calYear = parseInt(today.slice(0, 4));
let calMonth = parseInt(today.slice(5, 7));

function moveCal(direction) {
  calMonth += direction;
  if (calMonth > 12) { calMonth = 1; calYear++; }
  if (calMonth < 1)  { calMonth = 12; calYear--; }
  renderCal();
}

function goToday() {
  calYear = parseInt(today.slice(0, 4));
  calMonth = parseInt(today.slice(5, 7));
  renderCal();

### 3. 납기 상태 자동 판별
```javascript
const STATUS_META = {
  scheduled: { label: '예정',   color: '#64748b' },
  progress:  { label: '진행중', color: '#2563eb' },
  delay:     { label: '지연우려', color: '#d97706' },
  done:      { label: '납기완료', color: '#059669' },
  late:      { label: '납기지연', color: '#dc2626' },
};

// D-Day 계산
function daysLeft(dueDate) {
  return Math.ceil((new Date(dueDate) - new Date(today)) / 86400000);
}

// 자동 상태 판별 (상태를 수동 선택하지만 자동 제안도 가능)
function suggestStatus(item) {
  if (item.actualDate) return 'done';
  const dl = daysLeft(item.dueDate);
  if (dl < 0) return 'late';
  if (dl <= 3) return 'delay';
  return item.status;
}
```

### 4. 고객별 납기 준수율 계산
```javascript
function calcCustomerRates() {
  const customers = [...new Set(items.map(i => i.customer))];
  
  return customers.map(customer => {
    // 완료된 건만 대상 (예정/진행중은 아직 판단 불가)
    const completed = items.filter(i => 
      i.customer === customer && ['done', 'late'].includes(i.status)
    );
    const ontime = completed.filter(i => i.status === 'done').length;
    const pct = completed.length ? Math.round(ontime / completed.length * 100) : null;
    
    return { customer, pct, total: completed.length, ontime };
  }).filter(c => c.pct !== null).sort((a, b) => b.pct - a.pct);
}
```

### 미션 3 ⭐⭐⭐ — 납기 위험도 점수 계산
```javascript
// 각 납기 건의 위험도 점수 = 
//   (납기까지 남은 일수 / 전체 리드타임) × 100
// 점수가 낮을수록 위험 → 우선순위 하이라이팅
function calcRisk(item) {
  const totalDays = Math.ceil(
    (new Date(item.dueDate) - new Date(item.orderDate)) / 86400000
  );
  const remaining = Math.ceil(
    (new Date(item.dueDate) - new Date(today)) / 86400000
  );
  return totalDays ? Math.round(remaining / totalDays * 100) : 100;
}
```
