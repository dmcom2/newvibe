## 🎯 이 앱에서 배우는 핵심 개념

### 1. 날짜 기반 데이터 구조
```javascript
// records 배열: 각 레코드는 날짜+라인의 조합
let records = [
  { id: 1, date: '2026-06-01', line: '1호기', 
    produced: 850, good: 832, target: 900 },
  ...
]

// 특정 날짜의 데이터만 필터링
const dayRecords = records.filter(r => r.date === selectedDate);

// 월별 집계
const monthRecords = records.filter(r => r.date.startsWith('2026-06'));
```

### 2. 양품률 / 계획달성률 계산
```javascript
// 양품률 = 양품수 / 생산량 × 100
const yieldRate = r => r.produced ? (r.good / r.produced * 100) : 0;

// 계획달성률 = 생산량 / 목표량 × 100
const achieveRate = r => r.target ? (r.produced / r.target * 100) : 0;

// 전체 합계 양품률 (단순 평균이 아닌 가중 평균)
const totalYield = records.reduce((s,r) => s + r.good, 0) 
                 / records.reduce((s,r) => s + r.produced, 0) * 100;
```

### 6. 날짜 네비게이션
```javascript
// 이전/다음 날짜 이동
function changeDate(offset) {
  const d = new Date(selectedDate);
  d.setDate(d.getDate() + offset);
  selectedDate = d.toISOString().slice(0, 10);
  render();
}
```

### 7. CSV 내보내기 (한글 깨짐 방지 BOM)
```javascript
function exportCSV() {
  const rows = ['날짜,라인,목표,생산량,양품수,불량수,양품률'];
  records.forEach(r => {
    rows.push(`${r.date},${r.line},${r.target},${r.produced},
               ${r.good},${r.produced-r.good},${yieldRate(r).toFixed(1)}%`);
  });
  // ﻿ (BOM 문자) 앞에 붙여야 Excel에서 한글 안 깨짐
  const blob = new Blob(['﻿' + rows.join('\n')], 
                        { type: 'text/csv;charset=utf-8;' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = '생산실적.csv';
  a.click();
}
```
