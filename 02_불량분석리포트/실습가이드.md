### 2. 누적 비율 계산
```javascript
const DEFECT_TYPES = ['단락(Short)', '개방(Open)', '기공(Void)', '박리', '치수불량', '기타'];

function calcPareto(records) {
  // 1. 유형별 건수 집계
  const counts = {};
  DEFECT_TYPES.forEach(t => counts[t] = 0);
  records.forEach(r => {
    if (counts[r.type] !== undefined) counts[r.type] += r.count;
  });

  // 2. 내림차순 정렬
  const sorted = Object.entries(counts)
    .sort((a, b) => b[1] - a[1]);

  // 3. 누적 % 계산
  const total = sorted.reduce((s, [, v]) => s + v, 0);
  let cumSum = 0;
  return sorted.map(([type, count]) => {
    cumSum += count;
    return { type, count, cumPct: Math.round(cumSum / total * 100) };
  });
}
```

```javascript
// 불량 건수에 따른 자동 심각도 결정
function getSeverity(count, type) {
  if (type === '단락(Short)' || type === '개방(Open)') {
    if (count >= 10) return 'critical';
    if (count >= 3) return 'major';
  }
  return count >= 20 ? 'major' : 'minor';
}
```

### 색상 동적 생성 (불량 위험도 시각화)
```javascript
// 누적 80% 이하 항목은 빨간색, 이상은 회색
function getParetoColor(item, allItems) {
  const idx80 = allItems.findIndex(i => i.cumPct >= 80);
  const idx = allItems.indexOf(item);
  return idx <= idx80 
    ? 'rgba(220, 38, 38, 0.8)'   // 빨강 - 집중 개선 대상
    : 'rgba(156, 163, 175, 0.5)'; // 회색 - 일반 관리
}
```

```javascript
// 일별 데이터를 주차별로 자동 집계하는 함수를 만들어보세요
function groupByWeek(records) {
  const weekMap = {};
  records.forEach(r => {
    const date = new Date(r.date);
    const week = getWeekNumber(date);  // 주차 계산
    const key = `${date.getFullYear()}-W${week}`;
    if (!weekMap[key]) weekMap[key] = { count: 0, records: [] };
    weekMap[key].count += r.count;
    weekMap[key].records.push(r);
  });
  return weekMap;
}
```
