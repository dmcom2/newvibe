## 🎯 핵심 개념

### 1. 작업지시서 데이터 구조
```javascript
// WO (Work Order) 데이터
const wo = {
  id: 1001,
  no: 'WO-202606-001',          // 지시번호 (자동 채번)
  issueDate: '2026-06-01',      // 발행일
  customer: '(주)삼성전자',      // 고객사
  orderNo: 'PO-2026-001',       // 수주번호
  product: 'PCB-A2401',         // 제품명
  layer: '4층',                 // 층수
  qty: 1000,                    // 지시 수량
  produced: 650,                // 생산 완료 수량
  line: '1호기',                // 할당 라인
  startDate: '2026-06-03',      // 착수일
  dueDate: '2026-06-10',        // 납기일
  worker: '김생산',              // 담당자
  status: 'progress',           // 상태
  currentStep: '에칭',          // 현재 공정
  note: '',                     // 특이사항
};
```

### 2. 공정 단계 선택 UI (버튼 그룹)
```javascript
const PROCESS_STEPS = [
  '재단', '천공(드릴링)', '에칭', '도금', 
  '표면처리', '외관검사', '전기검사', '포장'
];

// 공정 단계 버튼 렌더링
function renderStepBtns(selected) {
  return PROCESS_STEPS.map(step => {
    const isActive = step === selected;
    return `<button 
      class="step-btn ${isActive ? 'active' : ''}" 
      onclick="selectStep('${step}')"
      style="${isActive ? 'background:var(--sky);color:#fff' : ''}"
    >${step}</button>`;
  }).join('');
}
```

### 3. 진행률 계산 및 시각화
```javascript
// 진행률 = 생산완료수 / 지시수량 × 100
const progPct = wo => wo.qty ? Math.round(wo.produced / wo.qty * 100) : 0;

// 진행률 바 HTML
`<div class="progress-bar">
  <div style="
    width: ${progPct(wo)}%;
    background: ${progPct(wo) >= 100 ? 'var(--green)' : 'var(--sky)'};
  "></div>
</div>`
```

### 4. 납기 D-Day 계산
```javascript
const today = new Date().toISOString().slice(0, 10);

function getDaysLeft(dueDate) {
  return Math.ceil((new Date(dueDate) - new Date(today)) / 86400000);
}

function isOverdue(wo) {
  return wo.status !== 'done' && wo.dueDate < today;
}

// 카드에서 납기 표시
const dl = getDaysLeft(wo.dueDate);
const dueText = isOverdue(wo) ? '⚠ 납기 지연' 
              : dl <= 3 && wo.status !== 'done' ? `D-${dl}` 
              : wo.dueDate;
```

```javascript
function generateWONo() {
  const today = new Date().toISOString().slice(0, 10);
  const y = today.slice(0, 4);
```

### 모달 편집 패턴 (공통 모달 재사용)
```javascript
// 신규 등록과 수정을 하나의 모달로 처리
let editId = null;

function openModal(id = null) {
  editId = id;  // null이면 신규, 있으면 수정
  const wo = id ? woList.find(x => String(x.id) === String(id)) : null;
  
  // 폼 초기화
  document.getElementById('w-product').value = wo?.product || '';
  // ...
  
  // 버튼 표시 조건
  document.getElementById('delBtn').style.display = id ? 'block' : 'none';
  document.getElementById('modal').classList.add('show');
}

function saveWO() {
  if (editId) {
    // 수정
    const idx = woList.findIndex(x => String(x.id) === String(editId));
    if (idx >= 0) woList[idx] = rec;
  } else {
    // 신규 추가
    woList.push(rec);
  }
}
```
