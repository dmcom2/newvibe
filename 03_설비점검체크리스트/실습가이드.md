### 2. 중첩 객체 안전 접근 (Optional Chaining)
```javascript
// 특정 날짜, 설비, 항목의 결과 가져오기
function getResult(date, eqId, itemId) {
  return results?.[date]?.[eqId]?.[itemId]?.result || null;
}

// 결과 저장
function setResult(date, eqId, itemId, result, note = '') {
  if (!results[date]) results[date] = {};
  if (!results[date][eqId]) results[date][eqId] = {};
  results[date][eqId][itemId] = { result, note };
  save();
}
```

### 3. 완료율 계산
```javascript
function calcCompletionRate(date, eqId) {
  const template = CHECK_TEMPLATES[eqId];
  const total = template.length;
  const done = template.filter(item => {
    const r = getResult(date, eqId, item.id);
    return r === 'ok' || r === 'ng';  // OK든 NG든 입력했으면 완료
  }).length;
  return total ? Math.round(done / total * 100) : 0;
}

// 전체 설비 완료율
const totalRate = Object.keys(CHECK_TEMPLATES).reduce((sum, eqId) => {
  return sum + calcCompletionRate(selectedDate, eqId);
}, 0) / Object.keys(CHECK_TEMPLATES).length;
```

### 4. 일괄 OK 처리
```javascript
// "전체 양호" 버튼: 해당 설비 모든 항목을 OK로 일괄 처리
function checkAll(eqId) {
  const template = CHECK_TEMPLATES[eqId];
  if (!results[selectedDate]) results[selectedDate] = {};
  if (!results[selectedDate][eqId]) results[selectedDate][eqId] = {};
  
  template.forEach(item => {
    // 이미 NG로 기록된 항목은 건드리지 않음
    if (results[selectedDate][eqId][item.id]?.result !== 'ng') {
      results[selectedDate][eqId][item.id] = { result: 'ok', note: '' };
    }
  });
  save();
  render();
}
```

### 5. 점검 주기별 필터
```javascript
// 일일/주간/월간 주기에 따라 오늘 해야 할 항목만 표시
const CYCLE_DAYS = { '일일': 1, '주간': 7, '월간': 30 };

function isScheduledToday(item, date) {
  if (item.cycle === '일일') return true;
  if (item.cycle === '주간') {
    // 월요일에만 주간 점검
    return new Date(date).getDay() === 1;
  }
  if (item.cycle === '월간') {
    // 매월 1일에만 월간 점검
    return new Date(date).getDate() === 1;
  }
  return false;
}
```
