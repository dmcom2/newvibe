// 자재 마스터 데이터 (기준 정보)
let items = [
  { id: 1, code: 'M001', cat: 'CCL', name: '동박적층판(CCL)',

### 2. 실시간 재고 계산 (트랜잭션 합산)
```javascript
// 현재 재고 = 초기재고 + 모든 입고 - 모든 출고
function calcStock(itemId) {
  const item = items.find(i => i.id === itemId);
  if (!item) return 0;

  const txTotal = txList
    .filter(tx => tx.itemId === itemId)
    .reduce((sum, tx) => {
      return sum + (tx.type === 'in' ? tx.qty : -tx.qty);
    }, 0);

  return item.stock + txTotal;
}

// 사용
const currentStock = calcStock(itemId);
```

### 3. 재고 상태 판별
```javascript
function getStatus(itemId) {
  const item = items.find(i => i.id === itemId);
  const stock = calcStock(itemId);

  if (stock <= item.min * 0.5) return 'low';    // 부족 (빨강)
  if (stock <= item.min) return 'warn';           // 주의 (주황)
  if (stock >= item.max) return 'over';           // 과잉 (파랑)
  return 'good';                                  // 정상 (초록)
}

const STATUS_META = {
  low:  { label: '부족',  color: '#dc2626', bg: '#fee2e2' },
  warn: { label: '주의',  color: '#d97706', bg: '#fef3c7' },
  good: { label: '정상',  color: '#059669', bg: '#dcfce7' },
  over: { label: '과잉',  color: '#2563eb', bg: '#dbeafe' },
};
```

### 4. 재고 수준 바 시각화
```javascript
function getStockBarWidth(itemId) {
  const item = items.find(i => i.id === itemId);
  const stock = calcStock(itemId);
  // max 기준 100%로 비율 계산
  return Math.min(100, Math.round(stock / item.max * 100));
}

// HTML
`<div class="stock-bar">
  <div style="
    width: ${getStockBarWidth(item.id)}%;
    background: ${STATUS_META[getStatus(item.id)].color};
    height: 100%;
  "></div>
</div>`
```

### 트랜잭션 패턴의 장점
```
장점:
✅ 이력 추적 완벽: 언제, 누가, 얼마나 입출고했는지 기록
✅ 롤백 가능: 잘못된 입력을 트랜잭션 삭제로 취소 가능
✅ 감사 추적: 재고 차이 발생시 이력으로 원인 파악

단점:
⚠ 데이터 증가: 트랜잭션이 쌓일수록 계산 시간 증가
   → 해결: 주기적으로 집계하여 초기재고 업데이트
```

### 성능 최적화: 재고 캐싱
```javascript
// 재고 계산을 매번 하면 느림 → 캐시 사용
let stockCache = {};

function invalidateCache() {
  stockCache = {};
}

function calcStockCached(itemId) {
  if (stockCache[itemId] !== undefined) return stockCache[itemId];
  stockCache[itemId] = calcStock(itemId);
  return stockCache[itemId];
}

// 데이터 저장 시 캐시 무효화
function save() {
  localStorage.setItem('pcbItems', JSON.stringify(items));
  localStorage.setItem('pcbTx', JSON.stringify(txList));
  invalidateCache();  // 캐시 초기화
}
```

// 자재별 월별 출고량을 집계하여 소비 추이 차트 표시
function getMonthlyConsumption(itemId, year) {
  const months = Array.from({length: 12}, (_, i) => i + 1);
  return months.map(m => {
    const monthStr = `${year}-${String(m).padStart(2, '0')}`;
    return txList
      .filter(tx => tx.itemId === itemId && tx.type === 'out' && tx.date.startsWith(monthStr))
      .reduce((sum, tx) => sum + tx.qty, 0);
  });
}
