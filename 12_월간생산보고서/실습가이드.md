### 4. 파레토 분석 자동 계산
```javascript
function calcPareto(defectCounts) {
  // 내림차순 정렬
  const sorted = [...defectCounts].sort((a, b) => b.cnt - a.cnt);
  
  const total = sorted.reduce((s, d) => s + d.cnt, 0);
  let cumSum = 0;
  
  return sorted.map(d => {
    cumSum += d.cnt;
    return {
      ...d,
      pct: Math.round(d.cnt / total * 100),        // 개별 비율
      cumPct: Math.round(cumSum / total * 100),     // 누적 비율
    };
  });
}

// 상위 80% 불량 유형 추출
const criticalDefects = defects.filter(d => d.cumPct <= 80);
```

### 미션 2 ⭐⭐ — 01_생산실적관리 데이터 연동
```javascript
// 실제 데이터를 읽어와서 보고서 생성
function getActualData(year, month) {
  const records = JSON.parse(localStorage.getItem('pcbRecords') || '[]');
  const monthStr = `${year}-${String(month).padStart(2, '0')}`;
  const monthRecords = records.filter(r => r.date.startsWith(monthStr));
  
  if (!monthRecords.length) {
    return null;  // 데이터 없음
  }
  
  // 일별 데이터 집계
  const dailyMap = {};
  monthRecords.forEach(r => {
    if (!dailyMap[r.date]) dailyMap[r.date] = { produced: 0, good: 0 };
    dailyMap[r.date].produced += r.produced;
    dailyMap[r.date].good += r.good;
  });
  
  return {
    dailyData: Object.entries(dailyMap).map(([date, v]) => ({
      day: parseInt(date.slice(8)),
      ...v,
      yieldRate: (v.good / v.produced * 100).toFixed(1),
    })).sort((a, b) => a.day - b.day),
    ...
  };
}
```
