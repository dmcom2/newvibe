### 2. 이수 이력 데이터 구조
```javascript
// 이수 이력: 직원ID + 교육ID + 날짜 (여러 번 이수 가능)
let completions = [
  { empId: 1001, courseId: 'C01', date: '2026-03-15', org: '사내', note: '' },
  { empId: 1001, courseId: 'C01', date: '2026-06-15', org: '사내', note: '' },  // 갱신
  { empId: 1002, courseId: 'C02', date: '2026-01-20', org: '외부기관', note: '' },
];

// 최근 이수 기록 가져오기
function getLastCompletion(empId, courseId) {
  return completions
    .filter(c => c.empId === empId && c.courseId === courseId)
    .sort((a, b) => b.date.localeCompare(a.date))[0] || null;
}
```

### 3. 이수 상태 계산 (핵심 로직)
```javascript
function getCellStatus(emp, course) {
  const last = getLastCompletion(emp.id, course.id);
  
  // 이수 이력 없음
  if (!last) return { status: 'none', text: '미이수', color: 'var(--mute)' };
  
  // 갱신 주기 없는 과목 (최초 1회)
  if (!course.cycleMonths) {
    return { status: 'ok', text: last.date.slice(0, 7), color: 'var(--green)' };
  }
  
  // 다음 이수 기한 계산
  const nextDue = new Date(last.date);
  nextDue.setMonth(nextDue.getMonth() + course.cycleMonths);
  
  const daysLeft = Math.ceil((nextDue - new Date(today)) / 86400000);
  
  if (daysLeft < 0)   return { status: 'overdue', text: '만료',   color: 'var(--red)' };
  if (daysLeft <= 30) return { status: 'due',     text: `D-${daysLeft}`, color: 'var(--amber)' };
  
  return { status: 'ok', text: last.date.slice(0, 7), color: 'var(--green)' };
}
```

### 4. 매트릭스(행렬) 렌더링
```javascript
function renderMatrix() {
  const emps = getFilteredEmps();
  
  // 헤더 행: 교육 과목명
  const header = COURSES.map(c => `
    <th style="border-left:2px solid ${c.color}40">
      ${c.name}
      <br><small>${c.required ? '(필수)' : '(선택)'}</small>
    </th>
  `).join('');
  
  // 데이터 행: 직원별 × 교육별 상태
  const rows = emps.map(emp => {
    const cells = COURSES.map(course => {
      const st = getCellStatus(emp, course);
      return `<td 
        style="color:${st.color};cursor:pointer"
        onclick="openCmModal('${emp.id}', '${course.id}')"
        title="${emp.name} - ${course.name}"
      >${st.text}</td>`;
    }).join('');
    
    return `<tr>
      <td onclick="openModal('emp', '${emp.id}')">${emp.name} (${emp.dept})</td>
      ${cells}
    </tr>`;
  }).join('');
  
  document.getElementById('matrixView').innerHTML = `
    <table><thead><tr><th>직원</th>${header}</tr></thead>
    <tbody>${rows}</tbody></table>`;
}
```

### 5. 이수율 계산
```javascript
// 전체 이수율: (이수완료 셀 수) / (전체 셀 수) × 100
function calcOverallRate() {
  const requiredCourses = COURSES.filter(c => c.required);
  const totalCells = emps.length * requiredCourses.length;
  const doneCells = emps.reduce((sum, emp) => {
    return sum + requiredCourses.filter(c => 
      getCellStatus(emp, c).status === 'ok'
    ).length;
  }, 0);
  
  return totalCells ? Math.round(doneCells / totalCells * 100) : 0;
}
```

## 🔍 코드 핵심 분석

### 법정교육 갱신 주기 관리
```
산업안전보건교육: 3개월 주기
  2026-01-15 이수 → 다음 이수 기한: 2026-04-15
  오늘(2026-06-30) → D-Day 계산: 만료됨!

소방안전교육: 12개월 주기
  2026-02-20 이수 → 다음 이수 기한: 2027-02-20
  오늘(2026-06-30) → D-Day 계산: D-235 (정상)
```

### 직원과 이수 이력을 별도로 저장하는 이유
```javascript
// ❌ 잘못된 방식: 직원 객체 안에 이수 이력 포함
const emp = {

  completions: { C01: '2026-03-15', C02: '2026-01-20' }
```

### 미션 3 ⭐⭐⭐ — 연간 교육 계획표 생성
```javascript
// 각 직원별로 올해 이수해야 할 교육 일정을 자동 생성
function genAnnualPlan(empId) {
  return COURSES.filter(c => c.required).flatMap(course => {
    if (course.cycleMonths === 0) return [];  // 최초 1회는 제외
    
    // 올해 해야 할 이수 횟수
    const timesPerYear = Math.floor(12 / course.cycleMonths);
    return Array.from({length: timesPerYear}, (_, i) => ({
      course: course.name,
      plannedMonth: (i + 1) * course.cycleMonths,
      done: false,
    }));
  });
}
```
