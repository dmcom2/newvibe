### 2. 실시간 금액 계산
```javascript
function calcRowTotal(idx) {
  const qty = parseFloat(document.getElementById(`i-qty-${idx}`)?.value) || 0;
  const price = parseFloat(document.getElementById(`i-price-${idx}`)?.value) || 0;
  const total = qty * price;
  
  const el = document.getElementById(`i-total-${idx}`);
  if (el) el.textContent = `₩${total.toLocaleString()}`;
  
  updateTotal();
}

function updateTotal() {
  let grandTotal = 0;
  // 모든 수량 × 단가 행 합산
  document.querySelectorAll('[id^="i-qty-"]').forEach(el => {
    const idx = el.id.split('-').pop();
    const qty = parseFloat(el.value) || 0;
    const price = parseFloat(document.getElementById(`i-price-${idx}`)?.value) || 0;
    grandTotal += qty * price;
  });
  document.getElementById('totalAmount').textContent = `₩${grandTotal.toLocaleString()}`;
}
```

```javascript
const po = {

  ]

// 입고 처리
function saveReceive() {
  const po = pos.find(x => String(x.id) === String(receivePoId));
  
  po.items.forEach((item, idx) => {
    const additionalQty = parseInt(document.getElementById(`r-qty-${idx}`)?.value) || 0;
    // 최대 발주량까지만 입고 가능
    item.received = Math.min(item.qty, item.received + additionalQty);
  });
  
  // 전체 입고 완료 여부 확인
  const allDone = po.items.every(i => i.received >= i.qty);
  po.status = allDone ? 'received' : 'partial';  // 입고완료 or 부분입고
}
```

### 5. 입고 진행률 표시
```javascript
function calcReceiveProgress(po) {
  const totalOrdered = po.items.reduce((s, i) => s + i.qty, 0);
  const totalReceived = po.items.reduce((s, i) => s + i.received, 0);
  return totalOrdered ? Math.round(totalReceived / totalOrdered * 100) : 0;
}
```

```javascript
// 업체별 평균 발주 주기 계산
function calcAvgOrderCycle(vendor) {
  const vendorPOs = pos
    .filter(p => p.vendor === vendor)
    .sort((a, b) => a.date.localeCompare(b.date));
  
  if (vendorPOs.length < 2) return null;
  
  let totalDays = 0;
  for (let i = 1; i < vendorPOs.length; i++) {
    totalDays += Math.ceil(
      (new Date(vendorPOs[i].date) - new Date(vendorPOs[i-1].date)) / 86400000
    );
  }
  return Math.round(totalDays / (vendorPOs.length - 1));
}
```

```javascript
function syncToInventory(po, receiveItems) {
  const txList = JSON.parse(localStorage.getItem('pcbTx') || '[]');
  receiveItems.forEach(item => {
    txList.push({
      id: Date.now(),
      date: today,
      type: 'in',
      itemName: item.name,
      qty: item.receivedNow,
      vendor: po.vendor,
      note: `발주서 ${po.no} 입고`,
    });
  });
  localStorage.setItem('pcbTx', JSON.stringify(txList));
}
```
